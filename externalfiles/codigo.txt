*!*							lntotal		= lntotal - lnimporte

					CASE INT(VAL(c_codigo)) = 551				&& PREMIOS
						lnpremios	= lnpremios + lnimporte

					CASE INT(VAL(c_codigo)) = 570				&& PREMIOS MI BINGO CATAMARQUEÑO
						lnpremios	= lnpremios + lnimporte

						lncomi		= lncomi + lnimporte
						lntotal		= lntotal - lnimporte

					CASE INT(VAL(c_codigo)) = 30				&& ANULACIÓN COMISIÓN
						lncomi		= lncomi - lnimporte

*!*							lntotal		= lntotal + lnimporte

					CASE INT(VAL(c_codigo)) = 23				&& FONDO DE GARANTIA
						lnfdogtia	= lnfdogtia + lnimporte
						lntotal		= lntotal + lnimporte

					CASE INT(VAL(c_codigo)) = 24				&& CAMARA DE AGENCIAS
						lndebitos	= lndebitos + lnimporte
						lntotal		= lntotal + lnimporte

					CASE INT(VAL(c_juego)) = 50 AND INT(VAL(c_codigo)) = 43		&& RECAUDACION
						lnrecauda	= lnrecauda + lnimporte
						lntotal		= lntotal + lnimporte
						lcjuego		= d_codigo
						lnmodal		= INT(VAL(c_codigo))

					OTHERWISE
						IF INT(VAL(c_codigo)) < 500
							lndebitos	= lndebitos + lnimporte
							lntotal		= lntotal + lnimporte
							lcjuego = d_codigo
						ELSE
							lncreditos	= lncreditos + lnimporte
							lntotal		= lntotal - lnimporte
							lcjuego = d_codigo
						ENDIF

				ENDCASE

			SKIP -1

			lck = PADL(lnagencia,4,"0")+"00"

			IF SEEK(lck,"maeagencias","agencia")

				INSERT INTO jij (cod_juego, modalidad, cjuego, no_sorteo, fecha_sort, no_agencia,;
					moneda, recaudacio, premios, multas, com_agenci, fdo_gtia, ing_brutos, municap, ncreditosv,;
					ndebitosv, TOTAL, fecha_vto, no_recibo, calta, falta) VALUES (lnjuego, lnmodal, lcjuego, lnsorteo,;
					CTOD(imp.d_operac), lnagencia, IIF(EMPTY(lcmoneda),"$",lcmoneda), lnrecauda, lnpremios*(-1),;
					0, lncomi, lnfdogtia, lningbru, 0, lncreditos, lndebitos, lntotal, TTOD(lfproc), lnrecibo, vg_username,;
					lfproc)

			ENDIF

		ENDSCAN

		USE IN imp

		SELECT jij

		SET RELATION TO PADL(no_agencia,4,'0')+'00' INTO maeagencias

		REPLACE ALL interior WITH maeagencias.interior, cod_agenci WITH maeagencias.cod_age, procesado WITH .F.
		lctitulo	= "TOTALES DE LIQUIDACIONES"
		SELECT jij

		SELECT cod_juego, modalidad, cjuego, no_sorteo, fecha_sort, no_agencia, moneda, SUM(recaudacio) AS recaudacio,;
			SUM(premios) AS premios, SUM(multas) AS multas, SUM(com_agenci) AS com_agenci, SUM(fdo_gtia) AS fdo_gtia,;
			SUM(ing_brutos) AS ing_brutos, SUM(municap) AS municap, SUM(ncreditosv) AS ncreditosv, SUM(ndebitosv) AS ndebitosv,;
			SUM(TOTAL) AS TOTAL, fecha_vto, no_recibo, calta, falta ;
			FROM jij ;
			ORDER BY cod_juego, no_sorteo ;
			GROUP BY cod_juego, no_sorteo ;
			INTO CURSOR jjcontrol READWRITE

		SELECT jjcontrol

		REPLACE ALL TOTAL WITH TOTAL + premios

		REPORT FORM reportes\rptctrlimpo TO PRINTER PROMPT PREVIEW NOCONSOLE

		USE IN jjcontrol

		IF 6 = MESSAGEBOX("El proceso de importación ha finalizado - Es correcto?",4+32,"¡ ATENCION !")

			IF INDEXSEEK(lfecproc,.T.,"paramfer","fecha")
				SELECT paramfer

				DO WHILE .T.
					SKIP -1
					IF domingo OR feriadon
						LOOP
					ENDIF
					EXIT
				ENDDO

				lfecproci	= fecha

			ELSE

				IF DOW(lfecproc - 1) = 7
					lfecproc	= lfecproc - 1
				ENDIF
				lfecproci	= lfecproc - 1

			ENDIF

			SELECT jij

			SCAN ALL

				SCATTER NAME oj
				lckey	= DTOS(fecha_sort)+PADL(no_sorteo,8,"0")+PADL(cod_juego,3,"0")+PADL(modalidad,2,"0")+PADL(no_agencia,8,"0")

				IF SEEK(lckey,"jjimpjue","id_mov")

					SELECT jjimpjue
					GATHER NAME oj
					SELECT jij

				ELSE

					INSERT INTO jjimpjue FROM NAME oj

				ENDIF

			ENDSCAN

			SELECT jjimpjue
			SET ORDER TO quinc
			SET FILTER TO fecha_sort = lfecproc

			SELECT jij
			ZAP

			SELECT jjimpjue
			GO TOP

			SCAN ALL
				SCATTER NAME oo
				INSERT INTO jij FROM NAME oo
			ENDSCAN

			SELECT jjimpjue
			SET FILTER TO
			SET ORDER TO quini
			SET FILTER TO fecha_sort = lfecproc		&&lfecproci
			GO TOP

			SCAN ALL
				SCATTER NAME oo
				INSERT INTO jij FROM NAME oo
			ENDSCAN

			SELECT cod_juego,modalidad,cjuego,no_sorteo,fecha_sort,cod_agenci,no_agencia,interior,moneda,SUM(recaudacio) AS recaudacio,;
				SUM(premios) AS premios,SUM(multas) AS multas,SUM(com_agenci) AS com_agenci,SUM(fdo_gtia) AS fdo_gtia,SUM(ing_brutos) AS ing_brutos,;
				SUM(municap) AS municap,SUM(ncreditosv) AS ncreditosv,SUM(ndebitosv) AS ndebitosv,SUM(TOTAL) AS TOTAL,fecha_vto,procesado ;
				FROM jij ;
				ORDER BY cod_juego, no_sorteo ;
				GROUP BY cod_juego, no_sorteo ;
				INTO CURSOR jjcontrol READWRITE

			REPLACE ALL TOTAL WITH TOTAL+premios
			GO TOP
			lctitulo	= "TOTALES DE LIQUIDACIONES CON INTERIOR"

				SCAN ALL

						TOTAL, fecha_vto, pagado, intereses, iva, total_gral, cajero, no_recibo, calta, falta, lscobro) VALUES;
						(jjimpjue.cod_juego, jjimpjue.modalidad, jjimpjue.cjuego, jjimpjue.no_sorteo, jjimpjue.fecha_sort,;
						jjimpjue.cod_agenci, jjimpjue.no_agencia, 0, jjimpjue.interior, jjimpjue.moneda,;
						jjimpjue.recaudacio, jjimpjue.premios, 0, jjimpjue.multas, jjimpjue.com_agenci, 0,;
						jjimpjue.fdo_gtia, jjimpjue.ing_brutos, jjimpjue.municap, jjimpjue.ncreditosv, jjimpjue.ndebitosv,;
						0, jjimpjue.TOTAL, jjimpjue.fecha_vto, jjimpjue.procesado, 0, 0, jjimpjue.TOTAL, "",;
						jjimpjue.no_recibo, jjimpjue.calta, jjimpjue.falta, .T.)

***************************************************

***************************************************
					IF lfec1 <= DATE() AND DATE() <= lfec2
						REPLACE lscim WITH .T. IN cajaliq

***************************************************

					REPLACE procesado WITH .T., falta WITH lfproc IN jjimpjue

						com_agencia, com_subage, fdo_gtia, ing_brutos, municap, ncreditosv, ndebitosv, t_com_subage,;

					INTO CURSOR tmpliq READWRITE

					SELECT tmpliq
					SCAN ALL
						IF fecha_vto < DATE() AND TOTAL > 0
							IF lscim
								REPLACE intereses	WITH 0,;
									iva				WITH 0,;
									total_gral		WITH TOTAL IN tmpliq
							ELSE
								vl_dias		= DATE()-fecha_vto
								vl_interes	= ROUND(TOTAL*vl_dias*vl_porint*.01,2)
								IF vl_interes < vl_minint
									vl_interes	= vl_minint
								ENDIF
								vl_iva		= ROUND(vl_interes*vl_poriva*.01,2)
								REPLACE intereses	WITH vl_interes,;
									iva				WITH vl_iva,;
									total_gral		WITH TOTAL+vl_interes+vl_iva IN tmpliq
							ENDIF
						ENDIF

						GROUP BY cod_agencia ;

				ENDIF

				IF 6 = MESSAGEBOX("Genera archivo para Transferencias InterBanking?",4+32,"Transferencias InterBanking")
					SUM ALL ABS(total_gral) TO lntransfer
					lnregistros = RECCOUNT()
					USE w:\egresos\lotesib IN 0

					INSERT INTO lotesib (corigen,falta,calta) VALUES ("CJ",DATETIME(),vg_username)
					lnlote = ID
					SELECT t_liq1
					lnh = FCREATE("W:\egresos\LotesIB\CJ"+PADL(lnlote,8,'0')+".txt")
					lnreg = 0
					lntra = 0
					SCAN ALL
						IF !EMPTY(cbu) AND !EMPTY(no_iva)
							creg = ""
							creg = "M"
							creg = creg+cbu
							creg = creg+STR(ABS(total_gral)*100,17,0)
							creg = creg+"FAC"
							creg = creg+PADR("PAGO PREMIO AGENCIA "+PADL(cod_agencia,6,'0'),60,' ')
							creg = creg+SPACE(22)
							creg = creg+SPACE(60)
							creg = creg+no_iva
							creg = creg+SPACE(42)
							IF FPUTS(lnh,creg) # 0
								lnreg = lnreg + 1
								lntra = lntra + ABS(total_gral)
							ENDIF
						ENDIF
					ENDSCAN
					= FCLOSE(lnh)
					IF lnreg # 0
						REPLACE nregistros WITH lnreg,;
							ntransfer WITH lntra,;
							lgenerado WITH .T. IN lotesib
					ELSE
						REPLACE lgenerado WITH .F. IN lotesib
					ENDIF
					USE IN lotesib
				ENDIF

				USE IN jjimpjue
				USE IN cajaliq

				SELECT jjimpjue
				DELETE ALL FOR falta = lfproc
				USE IN jjimpjue

				USE agjsjuegos!jjimpjue EXCLUSIVE IN 0

				SELECT jjimpjue
				PACK
				USE IN jjimpjue
				USE IN cajaliq

				= MESSAGEBOX("Ha cancelado la lectura de la liquidación de Juegos",0+64)

			ENDIF

		ENDIF

ENDCASE
